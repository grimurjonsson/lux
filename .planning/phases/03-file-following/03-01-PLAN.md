    ```
    if let Some(ref file_path) = cli.file {
        // FILE MODE -- but which sub-mode?
        if !std::io::stdin().is_terminal() {
            anyhow::bail!("cannot use file argument with piped input");
        }

        let has_explicit_n = cli.lines.is_some();
        let has_follow_flag = cli.follow_descriptor || cli.follow_name;

        // Three-mode detection:
        // 1. file + explicit -n + NO follow flag = PRINT AND EXIT
        // 2. file + follow flag (with or without -n) = FOLLOW
        // 3. bare file (no -n, no -f/-F) = implied -n 10 -F = FOLLOW
        let is_print_and_exit = has_explicit_n && !has_follow_flag;

        // Resolve line count: explicit -n value, or default 10 for follow mode
        let lines_str = cli.lines.as_deref().unwrap_or("10");
        let line_spec = tail::parse_line_spec(lines_str)?;

        let path = std::path::Path::new(file_path);
        let mut file = std::fs::File::open(path)
            .with_context(|| format!("cannot open '{file_path}'"))?;

        // Read initial lines
        let initial_lines = match line_spec {
            LineSpec::Last(n) => tail::read_last_n_lines(&mut file, n)?,
            LineSpec::FromLine(n) => tail::read_from_line(&mut file, n)?,
        };
        for line in &initial_lines {
            let output = engine.apply(line);
            writeln!(writer, "{output}")?;
        }
        writer.flush()?;

        if is_print_and_exit {
            // Print-and-exit mode: we already printed, just return
            return Ok(());
        }

        // Follow mode -- delegate to follow module (Plan 02 implements this)
        // For now, stub: TODO comment that Plan 02 will replace with follow::run(...)
        // IMPORTANT: keep file position at end after reading initial lines
        // Determine follow mode for Plan 02:

    } else {
        // Stdin pipe mode (existing behavior, unchanged)
        // -n is silently ignored per user decision
        let stdin = io::stdin().lock();

            let line = line?;
            let output = engine.apply(&line);
            writeln!(writer, "{output}")?;
        }
    }
    ```
