```rust
use regex::Regex;

pub struct LineFilter {
    include: Vec<Regex>,
    exclude: Vec<Regex>,
    strip_ansi: bool,
}

impl LineFilter {
    pub fn new(include: &[String], exclude: &[String], strip_ansi: bool) -> anyhow::Result<Self> {
        let include = include.iter()
            .map(|p| Regex::new(p).map_err(|e| anyhow::anyhow!("invalid include pattern '{}': {}", p, e)))
            .collect::<Result<Vec<_>, _>>()?;
        let exclude = exclude.iter()
            .map(|p| Regex::new(p).map_err(|e| anyhow::anyhow!("invalid exclude pattern '{}': {}", p, e)))
            .collect::<Result<Vec<_>, _>>()?;
        Ok(Self { include, exclude, strip_ansi })
    }

    pub fn is_active(&self) -> bool {
        !self.include.is_empty() || !self.exclude.is_empty()
    }

    pub fn should_show(&self, line: &str) -> bool {
        let matchable = if self.strip_ansi {
            crate::trigger::strip_ansi(line)
        } else {
            line.to_string()  // avoid allocation: use Cow if perf matters
        };
        let text = if self.strip_ansi { &matchable } else { line };

        let pass_include = self.include.is_empty()
            || self.include.iter().any(|r| r.is_match(text));
        let pass_exclude = self.exclude.iter().all(|r| !r.is_match(text));
        pass_include && pass_exclude
    }
}
```
